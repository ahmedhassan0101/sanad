import type { MembershipRole } from "@/db/models/Membership";

// Single source of truth for roles. No role checks outside can() + membership guards.
export const ROLES = ["center_admin", "branch_admin", "teacher"] as const;

export type Role = MembershipRole;

// Static matrix: can this role attempt this action at all (no DB needed).
// Stateful rules (creator protection, last admin) live in services/memberships.ts.
const MATRIX: Record<Role, Set<string>> = {
  center_admin: new Set([
    "branch:create",
    "member:invite",
    "member:change-role",
    "member:disable",
    "program:manage",
    "settings:manage",
  ]),
  branch_admin: new Set(["member:invite-teacher", "member:disable-teacher"]),
  teacher: new Set(["attendance:write", "progress:write"]),
};

export type Action =
  | "branch:create"
  | "member:invite"
  | "member:change-role"
  | "member:disable"
  | "member:invite-teacher"
  | "member:disable-teacher"
  | "program:manage"
  | "settings:manage"
  | "attendance:write"
  | "progress:write";

export function can(role: Role, action: Action): boolean {
  return MATRIX[role]?.has(action) ?? false;
}

// Rank for "no one grants above their own level".
export function rank(role: Role): number {
  return role === "center_admin" ? 3 : role === "branch_admin" ? 2 : 1;
}


import type { FilterQuery } from "mongoose";
import type { SessionContext } from "@/lib/tenant/context";

// One helper for every tenant read. orgId always comes from the server context.
export function scopedFilter(
  ctx: Extract<SessionContext, { kind: "member" }>,
  branchField = "branchId"
): FilterQuery<Record<string, unknown>> {
  const filter: FilterQuery<Record<string, unknown>> = { orgId: ctx.orgId };
  if (ctx.branchScope.mode === "list") {
    filter[branchField] = { $in: ctx.branchScope.branchIds };
  }
  return filter;
}



import "server-only";
import { Types } from "mongoose";
import { User } from "@/db/models/User";
import { Membership } from "@/db/models/Membership";
import type { SessionContext } from "@/lib/tenant/context";

// Builds context from the DB on every request. JWT carries userId only.
// Four states: platform (stats only) | onboarding (no org yet) | unverified | member.
export async function getSessionContext(
  userId: string | null,
  activeOrgId: string | null,
  expectedSessionVersion?: number | null,
): Promise<SessionContext> {
  if (!userId || !Types.ObjectId.isValid(userId))
    throw new Error("UNAUTHENTICATED");
  const user = await User.findById(userId).select(
    "status isPlatformAdmin sessionVersion email emailVerifiedAt",
  );
  if (!user || user.status !== "active") throw new Error("UNAUTHENTICATED");
  if (
    expectedSessionVersion !== undefined &&
    expectedSessionVersion !== user.sessionVersion
  ) {
    throw new Error("UNAUTHENTICATED");
  }
  if (user.isPlatformAdmin) return { kind: "platform", userId };
  if (!user.emailVerifiedAt)
    return { kind: "unverified", userId, email: user.email };
  if (!activeOrgId || !Types.ObjectId.isValid(activeOrgId))
    return { kind: "onboarding", userId };
  const membership = await Membership.findOne({
    userId: user._id,
    orgId: new Types.ObjectId(activeOrgId),
    status: "active",
  });
  if (!membership) throw new Error("FORBIDDEN");
  return {
    kind: "member",
    userId,
    orgId: membership.orgId,
    role: membership.role,
    branchScope:
      membership.role === "center_admin"
        ? { mode: "all" }
        : { mode: "list", branchIds: membership.branchIds },
  };
}




import "server-only";
import type { ClientSession, Types } from "mongoose";
import { Branch } from "@/db/models/Branch";
import { Membership, type MembershipRole } from "@/db/models/Membership";
import { Org } from "@/db/models/Org";
import { rank } from "@/lib/permissions/roles";

// Shape validation for every membership write (schema validators don't run on
// updateOne/findOneAndUpdate, so services enforce this instead).
export function assertMembershipShape(role: MembershipRole, branchIds: Types.ObjectId[]) {
  if ((role === "branch_admin" || role === "teacher") && branchIds.length < 1) {
    throw new Error("VALIDATION: scoped roles require at least one branch");
  }
}

// branchIds must be a subset of the org's branches. Called by membership services (2.2).
export async function assertBranchesInOrg(orgId: Types.ObjectId, branchIds: Types.ObjectId[]) {
  if (branchIds.length === 0) return;
  const count = await Branch.countDocuments({ _id: { $in: branchIds }, orgId });
  if (count !== branchIds.length) throw new Error("VALIDATION: branch outside org");
}

async function countActiveAdmins(
  orgId: Types.ObjectId,
  role: MembershipRole,
  branchId: Types.ObjectId | null,
  session?: ClientSession
) {
  const filter: Record<string, unknown> = { orgId, role, status: "active" };
  if (branchId) filter.branchIds = branchId;
  return Membership.countDocuments(filter).session(session ?? null);
}

// Stateful guard for demote/disable/remove. Static matrix stays in can();
// creator protection + last-admin need DB state, so they live here.
// NOTE (recorded improvement): full write-skew safety needs a shared write
// (Org.adminsVersion bump) inside the same transaction; count-check alone can
// race. Callers must run this inside a transaction in phase 5.
export async function guardAdminChange(
  orgId: Types.ObjectId,
  targetUserId: Types.ObjectId,
  targetRole: MembershipRole,
  targetBranchId: Types.ObjectId | null,
  session?: ClientSession
) {
  const org = await Org.findById(orgId).select("createdByUserId").session(session ?? null);
  if (org && org.createdByUserId.equals(targetUserId)) {
    throw new Error("FORBIDDEN: org creator is protected");
  }
  const remaining = await countActiveAdmins(orgId, targetRole, targetBranchId, session);
  if (remaining <= 1) throw new Error("FORBIDDEN: last admin at this level");
}

// Actor may not touch a higher role than their own.
export function assertRankAllows(actorRole: MembershipRole, targetRole: MembershipRole) {
  if (rank(actorRole) < rank(targetRole)) throw new Error("FORBIDDEN: above own level");
}



// إعداد الدخول: إيميل/باسورد + Google — بلا Adapter (حساباتنا في User).
import NextAuth, { type DefaultSession } from "next-auth";
import Credentials from "next-auth/providers/credentials";
import Google from "next-auth/providers/google";
import { z } from "zod";
import {
  ensureGoogleUser,
  getSessionVersionByUserId,
  resolveGoogleTokenUser,
  verifyCredentials,
} from "@/services/auth-accounts";

declare module "next-auth" {
  interface Session {
    user: {
      id: string;
      sessionVersion: number | null;
    } & DefaultSession["user"];
  }
}

const credentialsSchema = z.object({
  email: z.string().email(),
  password: z.string().min(1),
});

async function authorizeLogin(credentials: unknown) {
  const parsed = credentialsSchema.safeParse(credentials);
  if (!parsed.success) return null;
  return verifyCredentials(parsed.data.email, parsed.data.password);
}

const googleReady =
  !!process.env.GOOGLE_CLIENT_ID && !!process.env.GOOGLE_CLIENT_SECRET;

export const { handlers, auth, signIn, signOut } = NextAuth({
  session: { strategy: "jwt" },
  pages: { signIn: "/login" },
  providers: [
    ...(googleReady
      ? [
          Google({
            clientId: process.env.GOOGLE_CLIENT_ID,
            clientSecret: process.env.GOOGLE_CLIENT_SECRET,
          }),
        ]
      : []),
    Credentials({
      async authorize(credentials) {
        return authorizeLogin(credentials);
      },
    }),
  ],
  callbacks: {
    async jwt({ token, user, account }) {
      if (user?.id) {
        if (account?.provider === "google" && user.email) {
          const resolved = await resolveGoogleTokenUser(user.email);
          if (resolved) {
            (token as Record<string, unknown>).userId = resolved.id;
            (token as Record<string, unknown>).sessionVersion =
              resolved.sessionVersion;
          }
        } else {
          // Credentials ترجع Mongo _id مباشرة.
          (token as Record<string, unknown>).userId = user.id;
          const v = (user as { sessionVersion?: unknown }).sessionVersion;
          (token as Record<string, unknown>).sessionVersion =
            typeof v === "number"
              ? v
              : await getSessionVersionByUserId(user.id);
        }
      }
      return token;
    },
    async session({ session, token }) {
      const userId = (token as Record<string, unknown>).userId;
      if (typeof userId === "string") session.user.id = userId;
      const sv = (token as Record<string, unknown>).sessionVersion;
      session.user.sessionVersion = typeof sv === "number" ? sv : null;
      return session;
    },
    async signIn({ user, account }) {
      if (account?.provider === "google") {
        return ensureGoogleUser(user);
      }
      return true;
    },
    authorized({ auth: session, request }) {
      const publicPaths = [
        "/login",
        "/register",
        "/verify",
        "/forgot",
        "/reset",
      ];
      if (publicPaths.some((p) => request.nextUrl.pathname.startsWith(p)))
        return true;
      return !!session?.user;
    },
  },
});



// الدعوات: إنشاء كود لمرة واحدة، وقبوله بفحص ذري مع كل الصلاحيات.
import "server-only";
import { Types } from "mongoose";
import { connectDB } from "@/db/connection";
import { Branch } from "@/db/models/Branch";
import { Invite } from "@/db/models/Invite";
import { Membership, type MembershipRole } from "@/db/models/Membership";
import { generateInviteCode, hashInviteCode } from "@/lib/invite-code";
import { getEnv } from "@/lib/env";
import { can, rank } from "@/lib/permissions/roles";
import { requireVerifiedUser } from "./auth-accounts";
import { assertBranchesInOrg, assertMembershipShape } from "./memberships";

export async function createInvite(
  granter: { userId: string; orgId: Types.ObjectId; role: MembershipRole; branchIds: Types.ObjectId[] },
  role: MembershipRole,
  branchId: string | null,
  daysValid: number
) {
  await connectDB();
  await requireVerifiedUser(granter.userId);
  if (rank(granter.role) < rank(role)) throw new Error("FORBIDDEN: above own level");
  if (role === "center_admin" && !can(granter.role, "member:invite")) {
    throw new Error("FORBIDDEN");
  }
  const branchObj = branchId ? new Types.ObjectId(branchId) : null;
  if (role !== "center_admin" && !branchObj) throw new Error("VALIDATION: الفرع مطلوب");
  if (branchObj) {
    if (granter.role !== "center_admin" && !granter.branchIds.some((b) => b.equals(branchObj))) {
      throw new Error("FORBIDDEN: خارج فروعك");
    }
    await assertBranchesInOrg(granter.orgId, [branchObj]);
    const br = await Branch.findOne({ _id: branchObj, orgId: granter.orgId });
    if (!br || br.status !== "active") throw new Error("VALIDATION: الفرع غير نشط");
  }
  const activeCount = await Invite.countDocuments({
    orgId: granter.orgId,
    branchId: branchObj,
    usedAt: null,
    revokedAt: null,
    expiresAt: { $gt: new Date() },
  });
  if (activeCount >= 20) throw new Error("FORBIDDEN: وصلت لحد الدعوات النشطة");
  const code = generateInviteCode();
  const env = getEnv();
  const expiresAt = new Date(Date.now() + Math.min(Math.max(daysValid, 1), 7) * 86400_000);
  await Invite.create({
    orgId: granter.orgId,
    branchId: branchObj,
    role,
    codeHash: hashInviteCode(code, env.inviteCodePepper ?? "dev-pepper"),
    expiresAt,
    purgeAt: new Date(expiresAt.getTime() + 30 * 86400_000),
    createdByUserId: new Types.ObjectId(granter.userId),
  });
  return { code };
}

export async function consumeInvite(userId: string, rawCode: string) {
  await connectDB();
  await requireVerifiedUser(userId);
  const env = getEnv();
  const now = new Date();
  const updated = await Invite.findOneAndUpdate(
    {
      codeHash: hashInviteCode(rawCode, env.inviteCodePepper ?? "dev-pepper"),
      usedAt: null,
      revokedAt: null,
      expiresAt: { $gt: now },
    },
    { $set: { usedAt: now, usedByUserId: new Types.ObjectId(userId) } },
    { new: true }
  );
  if (!updated) throw new Error("الكود منتهي أو مستخدم أو غير صحيح");
  // Re-check granter authority at use time (not just at creation).
  const granter = await Membership.findOne({
    userId: updated.createdByUserId,
    orgId: updated.orgId,
    status: "active",
  });
  if (!granter || rank(granter.role) < rank(updated.role)) {
    throw new Error("الدعوة أُلغيت (تغيرت صلاحية الداعي)");
  }
  if (updated.branchId) {
    const br = await Branch.findOne({ _id: updated.branchId, orgId: updated.orgId });
    if (!br || br.status !== "active") throw new Error("الفرع غير نشط");
  }
  const uid = new Types.ObjectId(userId);
  const branchIds = updated.branchId ? [updated.branchId] : [];  assertMembershipShape(updated.role, branchIds as Types.ObjectId[]);
  const existing = await Membership.findOne({ userId: uid, orgId: updated.orgId });
  if (existing) {
    if (existing.status === "active") throw new Error("أنت عضو بالفعل في هذا المركز");
    existing.status = "active";
    existing.role = updated.role;
    existing.branchIds = branchIds as Types.ObjectId[];
    await existing.save();
  } else {
    await Membership.create({
      userId: uid,
      orgId: updated.orgId,
      role: updated.role,
      branchIds,
      invitedByUserId: updated.createdByUserId,
    });
  }
  return { orgId: String(updated.orgId) };
}

type TeamCtx = {
  kind: "member";
  orgId: Types.ObjectId;
  role: MembershipRole;
  branchScope: { mode: "all" } | { mode: "list"; branchIds: Types.ObjectId[] };
};

function requireTeamAccess(ctx: TeamCtx) {
  if (ctx.role !== "center_admin" && ctx.role !== "branch_admin") {
    throw new Error("FORBIDDEN: الفريق للمشرفين فقط");
  }
}

function branchFilter(ctx: TeamCtx) {
  if (ctx.branchScope.mode === "all") return {};
  return { branchId: { $in: ctx.branchScope.branchIds } };
}

// الدعوات الفعّالة فقط (غير مستخدمة وغير منتهية وغير ملغاة)، بالنطاق.
export async function listInvites(ctx: TeamCtx) {
  requireTeamAccess(ctx);
  await connectDB();
  const invites = await Invite.find({
    orgId: ctx.orgId,
    ...branchFilter(ctx),
    usedAt: null,
    revokedAt: null,
    expiresAt: { $gt: new Date() },
  })
    .sort({ createdAt: -1 })
    .lean();
  return invites.map((i) => ({
    id: String(i._id),
    role: i.role,
    branchId: i.branchId ? String(i.branchId) : null,
    expiresAt: i.expiresAt.toISOString(),
  }));
}

export async function revokeInvite(ctx: TeamCtx, inviteId: string) {
  requireTeamAccess(ctx);
  if (!Types.ObjectId.isValid(inviteId)) throw new Error("FORBIDDEN");
  await connectDB();
  const invite = await Invite.findOne({ _id: new Types.ObjectId(inviteId), orgId: ctx.orgId });
  if (!invite || invite.usedAt || invite.revokedAt) throw new Error("NOT_FOUND");
  if (ctx.role === "branch_admin") {
    const mine = ctx.branchScope.mode === "list" ? ctx.branchScope.branchIds : [];
    if (!invite.branchId || !mine.some((b) => b.equals(invite.branchId))) {
      throw new Error("FORBIDDEN: خارج فروعك");
    }
  }
  invite.revokedAt = new Date();
  await invite.save();
  return { ok: true as const };
}
