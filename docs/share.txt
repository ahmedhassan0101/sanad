const ROLE_NAMES: Record<string, string> = {
  center_admin: "مدير مركز",
  branch_admin: "مشرف فرع",
  teacher: "معلم",
};

export default async function TeamPage({
  searchParams,
}: {
  searchParams: Promise<{ created?: string; error?: string }>;
}) {
  
  const ctx = await viewer();
  if (ctx.kind !== "member") redirect("/verify-pending");
  const sp = await searchParams;
  const data = await teamData().catch(() => null);
  if (!data) redirect("/dashboard");
  const roleOptions =
    data.role === "center_admin"
      ? ["center_admin", "branch_admin", "teacher"]
      : ["branch_admin", "teacher"];
  const branchName = new Map(data.branches.map((b) => [b.id, b.name]));
  return (
    <div className="flex flex-col gap-4">
      <h1 className="text-xl font-bold">الفريق</h1>
      {sp.error && (
        <p className="text-sm text-destructive">
          تعذر تنفيذ العملية — تحقق من الصلاحيات
        </p>
      )}
      {sp.created && (
        <Card>
          <CardHeader>
            <CardTitle>الكود (يظهر مرة واحدة)</CardTitle>
            <CardDescription className="text-lg font-bold" dir="ltr">
              {sp.created}
            </CardDescription>
          </CardHeader>
          <CardContent>
            <CopyInvite code={sp.created} />
          </CardContent>
        </Card>
      )}
      <Card>
        <CardHeader>
          <CardTitle>دعوة جديدة</CardTitle>
        </CardHeader>
        <CardContent>
          <form action={createInviteAction} className="flex flex-col gap-3">
            <label className="text-sm">
              الدور
              <select
                name="role"
                className="mt-1 w-full rounded-md border p-2"
                defaultValue="teacher"
              >
                {roleOptions.map((r) => (
                  <option key={r} value={r}>
                    {ROLE_NAMES[r]}
                  </option>
                ))}
              </select>
            </label>
            {data.branches.length > 1 ? (
              <label className="text-sm">
                الفرع
                <select
                  name="branchId"
                  className="mt-1 w-full rounded-md border p-2"
                >
                  {data.branches.map((b) => (
                    <option key={b.id} value={b.id}>
                      {b.name}
                    </option>
                  ))}
                </select>
              </label>
            ) : (
              <input
                type="hidden"
                name="branchId"
                value={data.branches[0]?.id ?? ""}
              />
            )}
            <label className="text-sm">
              المدة بالأيام (حتى 7)
              <input
                name="days"
                type="number"
                min={1}
                max={7}
                defaultValue={2}
                className="mt-1 w-full rounded-md border p-2"
                dir="ltr"
              />
            </label>
            <Button type="submit">إنشاء الكود</Button>
          </form>
        </CardContent>
      </Card>
      <Card>
        <CardHeader>
          <CardTitle>الدعوات الفعّالة</CardTitle>
        </CardHeader>
        <CardContent className="flex flex-col gap-2">
          {data.invites.length === 0 && (
            <p className="text-sm text-muted-foreground">لا دعوات فعّالة</p>
          )}
          {data.invites.map((i) => (
            <form
              key={i.id}
              action={revokeInviteAction.bind(null, i.id)}
              className="flex items-center justify-between gap-2 text-sm"
            >
              <span>
                {ROLE_NAMES[i.role]} —{" "}
                {i.branchId ? (branchName.get(i.branchId) ?? "") : "كل الفروع"}
              </span>
              <Button type="submit" variant="outline" size="sm">
                إلغاء
              </Button>
            </form>
          ))}
        </CardContent>
      </Card>
      <Card>
        <CardHeader>
          <CardTitle>الأعضاء</CardTitle>
        </CardHeader>
        <CardContent className="flex flex-col gap-3">
          {data.members.map((m) => (
            <div
              key={m.userId}
              className="flex flex-col gap-1 border-b pb-2 text-sm"
            >
              <span>
                {m.name} — {m.email} — {ROLE_NAMES[m.role]} —{" "}
                {m.status === "active" ? "مفعّل" : "معطّل"}
              </span>
              <div className="flex gap-2">
                <form
                  action={updateMemberAction.bind(null, m.userId)}
                  className="flex gap-2"
                >
                  <select
                    name="role"
                    defaultValue={m.role}
                    className="rounded-md border p-1"
                  >
                    {roleOptions.map((r) => (
                      <option key={r} value={r}>
                        {ROLE_NAMES[r]}
                      </option>
                    ))}
                  </select>
                  <Button type="submit" variant="outline" size="sm">
                    تغيير الدور
                  </Button>
                </form>
                <form action={updateMemberAction.bind(null, m.userId)}>
                  <input
                    type="hidden"
                    name="status"
                    value={m.status === "active" ? "disabled" : "active"}
                  />
                  <Button type="submit" variant="outline" size="sm">
                    {m.status === "active" ? "تعطيل" : "تفعيل"}
                  </Button>
                </form>
              </div>
            </div>
          ))}
        </CardContent>
      </Card>
    </div>
  );
}

import "server-only";
import { cookies } from "next/headers";
import { auth } from "@/lib/auth/config";
import { getSessionContext } from "./session-context";

export async function viewer() {
  const session = await auth();
  if (!session?.user?.id) return { kind: "logged-out" } as const;
  const jar = await cookies();
  const activeOrg = jar.get("active_org")?.value ?? null;
  return getSessionContext(
    session.user.id,
    activeOrg,
    session.user.sessionVersion ?? null,
  );
}

// Builds context from the DB on every request. JWT carries userId only.
// Four states: platform (stats only) | onboarding (no org yet) | unverified | member.
export async function getSessionContext(
  userId: string | null,
  activeOrgId: string | null,
  expectedSessionVersion?: number | null,
): Promise<SessionContext> {
  if (!userId || !Types.ObjectId.isValid(userId))
    throw new Error("UNAUTHENTICATED");
  const user = await User.findById(userId).select(
    "status isPlatformAdmin sessionVersion email emailVerifiedAt",
  );
  if (!user || user.status !== "active") throw new Error("UNAUTHENTICATED");
  if (
    expectedSessionVersion !== undefined &&
    expectedSessionVersion !== user.sessionVersion
  ) {
    throw new Error("UNAUTHENTICATED");
  }
  if (user.isPlatformAdmin) return { kind: "platform", userId };
  if (!user.emailVerifiedAt)
    return { kind: "unverified", userId, email: user.email };
  if (!activeOrgId || !Types.ObjectId.isValid(activeOrgId))
    return { kind: "onboarding", userId };
  const membership = await Membership.findOne({
    userId: user._id,
    orgId: new Types.ObjectId(activeOrgId),
    status: "active",
  });
  if (!membership) throw new Error("FORBIDDEN");
  return {
    kind: "member",
    userId,
    orgId: membership.orgId,
    role: membership.role,
    branchScope:
      membership.role === "center_admin"
        ? { mode: "all" }
        : { mode: "list", branchIds: membership.branchIds },
  };
}

import type { Types } from "mongoose";
import type { MembershipRole } from "@/db/models/Membership";

// Pure types only — no DB imports allowed in lib.
export type BranchScope =
  | { mode: "all" }
  | { mode: "list"; branchIds: Types.ObjectId[] };

export type SessionContext =
  | { kind: "platform"; userId: string }
  | { kind: "onboarding"; userId: string }
  | { kind: "unverified"; userId: string; email: string }
  | {
      kind: "member";
      userId: string;
      orgId: Types.ObjectId;
      role: MembershipRole;
      branchScope: BranchScope;
    };



export async function teamData() {
  const ctx = await teamCtx();
  const pref = (await cookies()).get("active_branch")?.value ?? null;
  const [branches, members, invites] = await Promise.all([
    listBranches(ctx),
    listMembers(ctx),
    listInvites(ctx),
  ]);
  const validPref = branches.some((b) => String(b._id) === pref) ? pref : null;
  return {
    role: ctx.role,
    branches: branches.map((b) => ({ id: String(b._id), name: b.name })),
    members: validPref ? members.filter((m) => m.branchIds.includes(validPref)) : members,
    invites: validPref ? invites.filter((i) => !i.branchId || i.branchId === validPref) : invites,
  };
}

async function teamCtx() {
  const ctx = await viewer();
  if (ctx.kind !== "member") redirect("/verify-pending");
  if (ctx.role !== "center_admin" && ctx.role !== "branch_admin")
    redirect("/dashboard");
  return ctx;
}

// Minimal witness service for isolation tests: lists branches inside the caller's scope.
// Branch identity IS _id here, so scope filters on _id (not branchId).
export async function listBranches(ctx: SessionContext) {
  if (ctx.kind !== "member") throw new Error("FORBIDDEN");
  return Branch.find(scopedFilter(ctx, "_id")).sort({ name: 1 }).lean();
}



export async function listMembers(ctx: MemberCtx) {
  requireTeamAccess(ctx);
  await connectDB();
  const memberships = await Membership.find(scopedFilter(ctx, "branchIds")).sort({ createdAt: 1 }).lean();
  const users = await User.find({ _id: { $in: memberships.map((m) => m.userId) } })
    .select("name email")
    .lean();
  const nameOf = new Map(users.map((u) => [String(u._id), { name: u.name, email: u.email }]));
  return memberships.map((m) => ({
    userId: String(m.userId),
    ...(nameOf.get(String(m.userId)) ?? { name: "", email: "" }),
    role: m.role,
    branchIds: m.branchIds.map(String),
    status: m.status,
  }));
}



// الدعوات الفعّالة فقط (غير مستخدمة وغير منتهية وغير ملغاة)، بالنطاق.
export async function listInvites(ctx: TeamCtx) {
  requireTeamAccess(ctx);
  await connectDB();
  const invites = await Invite.find({
    orgId: ctx.orgId,
    ...branchFilter(ctx),
    usedAt: null,
    revokedAt: null,
    expiresAt: { $gt: new Date() },
  })
    .sort({ createdAt: -1 })
    .lean();
  return invites.map((i) => ({
    id: String(i._id),
    role: i.role,
    branchId: i.branchId ? String(i.branchId) : null,
    expiresAt: i.expiresAt.toISOString(),
  }));
}



// أكشنات الفريق: إنشاء/إلغاء دعوة + تغيير دور/تعطيل عضو — كلها بفحص سيرفر.
"use server";

import { cookies } from "next/headers";
import { redirect } from "next/navigation";
import { createInvite, listInvites, revokeInvite } from "@/services/invites";
import { listBranches } from "@/services/branches-read";
import { listMembers, updateMember } from "@/services/members";
import { viewer } from "@/services/viewer";
import { inviteFormSchema } from "@/validators/forms";

async function teamCtx() {
  const ctx = await viewer();
  if (ctx.kind !== "member") redirect("/verify-pending");
  if (ctx.role !== "center_admin" && ctx.role !== "branch_admin")
    redirect("/dashboard");
  return ctx;
}

export async function createInviteAction(formData: FormData) {
  const ctx = await teamCtx();
  const parsed = inviteFormSchema.safeParse({
    role: String(formData.get("role") ?? ""),
    branchId: formData.get("branchId")
      ? String(formData.get("branchId"))
      : null,
    days: Number(formData.get("days") ?? 2),
  });
  if (!parsed.success) redirect("/dashboard/team?error=1");
  try {
    const { code } = await createInvite(
      {
        userId: ctx.userId,
        orgId: ctx.orgId,
        role: ctx.role,
        branchIds:
          ctx.branchScope.mode === "list" ? ctx.branchScope.branchIds : [],
      },
      parsed.data.role,
      parsed.data.branchId ?? null,
      parsed.data.days,
    );
    redirect(`/dashboard/team?created=${code}`);
  } catch {
    redirect("/dashboard/team?error=1");
  }
}

export async function revokeInviteAction(inviteId: string) {
  const ctx = await teamCtx();
  try {
    await revokeInvite(ctx, inviteId);
  } catch {
    redirect("/dashboard/team?error=1");
  }
  redirect("/dashboard/team");
}

export async function updateMemberAction(
  targetUserId: string,
  formData: FormData,
) {
  const ctx = await teamCtx();
  const role = formData.get("role");
  const status = formData.get("status");
  try {
    await updateMember(ctx, targetUserId, {
      ...(role
        ? { role: String(role) as "center_admin" | "branch_admin" | "teacher" }
        : {}),
      ...(status ? { status: String(status) as "active" | "disabled" } : {}),
    });
  } catch (e) {
    redirect(
      `/dashboard/team?error=${encodeURIComponent(e instanceof Error ? e.message : "1")}`,
    );
  }
  redirect("/dashboard/team");
}

export async function teamData() {
  const ctx = await teamCtx();
  const pref = (await cookies()).get("active_branch")?.value ?? null;
  const [branches, members, invites] = await Promise.all([
    listBranches(ctx),
    listMembers(ctx),
    listInvites(ctx),
  ]);
  const validPref = branches.some((b) => String(b._id) === pref) ? pref : null;
  return {
    role: ctx.role,
    branches: branches.map((b) => ({ id: String(b._id), name: b.name })),
    members: validPref
      ? members.filter((m) => m.branchIds.includes(validPref))
      : members,
    invites: validPref
      ? invites.filter((i) => !i.branchId || i.branchId === validPref)
      : invites,
  };
}
